// MyArray - динамический массив строк с удвоением ёмкости.
type MyArray struct {
        data []string
        size int
}

// NewMyArray создаёт массив с начальной ёмкостью capacity (минимум 1).
func NewMyArray(capacity int) *MyArray {
        if capacity < 1 {
                capacity = 1
        }
        return &MyArray{data: make([]string, capacity)}
}

func (a *MyArray) TypeName() string { return "MyArray" }
func (a *MyArray) Size() int        { return a.size }
func (a *MyArray) Capacity() int    { return len(a.data) }
func (a *MyArray) Clear()           { a.size = 0 }

func (a *MyArray) grow() {
        if a.size < len(a.data) {
                return
        }
        n := len(a.data) * 2
        if n == 0 {
                n = 2
        }
        fresh := make([]string, n)
        copy(fresh, a.data[:a.size])
        a.data = fresh
}

// PushBack добавляет элемент в конец.
func (a *MyArray) PushBack(v string) {
        a.grow()
        a.data[a.size] = v
        a.size++
}

// InsertAt вставляет элемент по индексу (index == Size() допустим).
func (a *MyArray) InsertAt(index int, v string) error {
        if index < 0 || index > a.size {
                return ErrIndex
        }
        a.grow()
        copy(a.data[index+1:a.size+1], a.data[index:a.size])
        a.data[index] = v
        a.size++
        return nil
}

// GetAt возвращает элемент по индексу.
func (a *MyArray) GetAt(index int) (string, error) {
        if index < 0 || index >= a.size {
                return "", ErrIndex
        }
        return a.data[index], nil
}

// SetAt заменяет элемент по индексу.
func (a *MyArray) SetAt(index int, v string) error {
        if index < 0 || index >= a.size {
                return ErrIndex
        }
        a.data[index] = v
        return nil
}

// RemoveAt удаляет элемент по индексу.
func (a *MyArray) RemoveAt(index int) error {
        if index < 0 || index >= a.size {
                return ErrIndex
        }
        copy(a.data[index:], a.data[index+1:a.size])
        a.size--
        a.data[a.size] = ""
        return nil
}

func (a *MyArray) ToSlice() []string {
        out := make([]string, a.size)
        copy(out, a.data[:a.size])
        return out
}

func (a *MyArray) Append(v string) error {
        a.PushBack(v)
        return nil
}

// Package ds содержит структуры данных ЛР3 (Go-версия) и их сериализацию.
//
// Форматы файлов совпадают с C++-версией:
//
//        бинарный: "LR3B" | u8 len(typeName) | typeName | u32 count | count * (u32 len | bytes)  (little-endian)
//        текстовый: "LR3T <typeName> <count>\n" и по одной строке на элемент (\\ \n \r экранируются)
package ds

import (
        "bufio"
        "bytes"
        "encoding/binary"
        "errors"
        "fmt"
        "io"
        "strconv"
        "strings"
)

// ErrEmpty возвращается при чтении из пустой структуры.
var ErrEmpty = errors.New("ds: container is empty")

// ErrIndex возвращается при обращении по индексу вне диапазона.
var ErrIndex = errors.New("ds: index out of range")

const (
        magicBin  = "LR3B"
        magicText = "LR3T"
        maxLen    = 64 * 1024 * 1024
)

// Container - общий интерфейс всех структур; сериализация строится на его пяти методах.
type Container interface {
        TypeName() string
        Size() int
        ToSlice() []string // порядок, достаточный для восстановления
        Clear()
        Append(value string) error // добавление при десериализации
}

// SaveBinary записывает контейнер в бинарном формате.
func SaveBinary(c Container, w io.Writer) error {
        name := c.TypeName()
        items := c.ToSlice()
        var buf bytes.Buffer
        buf.WriteString(magicBin)
        buf.WriteByte(byte(len(name)))
        buf.WriteString(name)
        var u [4]byte
        binary.LittleEndian.PutUint32(u[:], uint32(len(items)))
        buf.Write(u[:])
        for _, s := range items {
                binary.LittleEndian.PutUint32(u[:], uint32(len(s)))
                buf.Write(u[:])
                buf.WriteString(s)
        }
        _, err := w.Write(buf.Bytes())
        return err
}

// LoadBinary заменяет содержимое контейнера данными из бинарного потока.
// При ошибке контейнер не изменяется.
func LoadBinary(c Container, r io.Reader) error {
        magic := make([]byte, 4)
        if _, err := io.ReadFull(r, magic); err != nil || string(magic) != magicBin {
                return errors.New("ds: bad binary magic")
        }
        var nameLen [1]byte
        if _, err := io.ReadFull(r, nameLen[:]); err != nil {
                return fmt.Errorf("ds: unexpected end of binary data: %w", err)
        }
        name := make([]byte, nameLen[0])
        if _, err := io.ReadFull(r, name); err != nil {
                return fmt.Errorf("ds: unexpected end of binary data: %w", err)
        }
        if string(name) != c.TypeName() {
                return fmt.Errorf("ds: type mismatch: file has %s, expected %s", name, c.TypeName())
        }
        count, err := readU32(r)
        if err != nil {
                return err
        }
        var items []string
        for i := uint32(0); i < count; i++ {
                n, err := readU32(r)
                if err != nil {
                        return err
                }
                if n > maxLen {
                        return errors.New("ds: element too large")
                }
                b := make([]byte, n)
                if _, err := io.ReadFull(r, b); err != nil {
                        return fmt.Errorf("ds: unexpected end of binary data: %w", err)
                }
                items = append(items, string(b))
        }
        return replace(c, items)
}

func readU32(r io.Reader) (uint32, error) {
        var b [4]byte
        if _, err := io.ReadFull(r, b[:]); err != nil {
                return 0, fmt.Errorf("ds: unexpected end of binary data: %w", err)
        }
        return binary.LittleEndian.Uint32(b[:]), nil
}

// SaveText записывает контейнер в текстовом формате.
func SaveText(c Container, w io.Writer) error {
        items := c.ToSlice()
        var sb strings.Builder
        fmt.Fprintf(&sb, "%s %s %d\n", magicText, c.TypeName(), len(items))
        for _, s := range items {
                sb.WriteString(escape(s))
                sb.WriteByte('\n')
        }
        _, err := io.WriteString(w, sb.String())
        return err
}

// LoadText заменяет содержимое контейнера данными из текстового потока.
func LoadText(c Container, r io.Reader) error {
        br := bufio.NewReader(r)
        header, err := br.ReadString('\n')
        if err != nil && header == "" {
                return errors.New("ds: bad text header")
        }
        fields := strings.Fields(header)
        if len(fields) != 3 || fields[0] != magicText {
                return errors.New("ds: bad text header")
        }
        if fields[1] != c.TypeName() {
                return fmt.Errorf("ds: type mismatch: file has %s, expected %s", fields[1], c.TypeName())
        }
        count, err := strconv.Atoi(fields[2])
        if err != nil || count < 0 {
                return errors.New("ds: bad text header")
        }
        var items []string
        for i := 0; i < count; i++ {
                line, err := br.ReadString('\n')
                if err != nil {
                        return errors.New("ds: unexpected end of text data")
                }
                s, err := unescape(strings.TrimSuffix(line, "\n"))
                if err != nil {
                        return err
                }
                items = append(items, s)
        }
        return replace(c, items)
}

// replace атомарно (с откатом) подменяет содержимое контейнера.
func replace(c Container, items []string) error {
        backup := c.ToSlice()
        c.Clear()
        for _, s := range items {
                if err := c.Append(s); err != nil {
                        c.Clear()
                        for _, old := range backup {
                                _ = c.Append(old)
                        }
                        return err
                }
        }
        return nil
}

func escape(s string) string {
        r := strings.NewReplacer(`\`, `\\`, "\n", `\n`, "\r", `\r`)
        return r.Replace(s)
}

func unescape(s string) (string, error) {
        var sb strings.Builder
        for i := 0; i < len(s); i++ {
                if s[i] != '\\' {
                        sb.WriteByte(s[i])
                        continue
                }
                i++
                if i >= len(s) {
                        return "", errors.New("ds: dangling escape in text data")
                }
                switch s[i] {
                case '\\':
                        sb.WriteByte('\\')
                case 'n':
                        sb.WriteByte('\n')
                case 'r':
                        sb.WriteByte('\r')
                default:
                        return "", errors.New("ds: unknown escape in text data")
                }
        }
        return sb.String(), nil
}

// Format выводит содержимое в виде "[a, b, c]".
func Format(c Container) string {
        return "[" + strings.Join(c.ToSlice(), ", ") + "]"
}

type doublyNode struct {
        value      string
        next, prev *doublyNode
}

// DoublyLinkedList - двусвязный список.
type DoublyLinkedList struct {
        head, tail *doublyNode
        size       int
}

func (l *DoublyLinkedList) TypeName() string { return "DoublyLinkedList" }
func (l *DoublyLinkedList) Size() int        { return l.size }
func (l *DoublyLinkedList) Clear()           { l.head, l.tail, l.size = nil, nil, 0 }

func (l *DoublyLinkedList) PushFront(v string) {
        n := &doublyNode{value: v, next: l.head}
        if l.head != nil {
                l.head.prev = n
        }
        l.head = n
        if l.tail == nil {
                l.tail = n
        }
        l.size++
}

func (l *DoublyLinkedList) PushBack(v string) {
        n := &doublyNode{value: v, prev: l.tail}
        if l.tail != nil {
                l.tail.next = n
        }
        l.tail = n
        if l.head == nil {
                l.head = n
        }
        l.size++
}

func (l *DoublyLinkedList) find(v string) *doublyNode {
        for n := l.head; n != nil; n = n.next {
                if n.value == v {
                        return n
                }
        }
        return nil
}

func (l *DoublyLinkedList) unlink(n *doublyNode) {
        if n.prev != nil {
                n.prev.next = n.next
        } else {
                l.head = n.next
        }
        if n.next != nil {
                n.next.prev = n.prev
        } else {
                l.tail = n.prev
        }
        l.size--
}

func (l *DoublyLinkedList) Contains(v string) bool { return l.find(v) != nil }

func (l *DoublyLinkedList) InsertAfter(target, value string) bool {
        t := l.find(target)
        if t == nil {
                return false
        }
        n := &doublyNode{value: value, prev: t, next: t.next}
        if t.next != nil {
                t.next.prev = n
        } else {
                l.tail = n
        }
        t.next = n
        l.size++
        return true
}

func (l *DoublyLinkedList) InsertBefore(target, value string) bool {
        t := l.find(target)
        if t == nil {
                return false
        }
        n := &doublyNode{value: value, next: t, prev: t.prev}
        if t.prev != nil {
                t.prev.next = n
        } else {
                l.head = n
        }
        t.prev = n
        l.size++
        return true
}

func (l *DoublyLinkedList) PopFront() (string, error) {
        if l.head == nil {
                return "", ErrEmpty
        }
        v := l.head.value
        l.unlink(l.head)
        return v, nil
}

func (l *DoublyLinkedList) PopBack() (string, error) {
        if l.tail == nil {
                return "", ErrEmpty
        }
        v := l.tail.value
        l.unlink(l.tail)
        return v, nil
}

func (l *DoublyLinkedList) RemoveByValue(v string) bool {
        n := l.find(v)
        if n == nil {
                return false
        }
        l.unlink(n)
        return true
}

func (l *DoublyLinkedList) RemoveAfter(target string) bool {
        t := l.find(target)
        if t == nil || t.next == nil {
                return false
        }
        l.unlink(t.next)
        return true
}

func (l *DoublyLinkedList) RemoveBefore(target string) bool {
        t := l.find(target)
        if t == nil || t.prev == nil {
                return false
        }
        l.unlink(t.prev)
        return true
}

func (l *DoublyLinkedList) GetAt(index int) (string, error) {
        if index < 0 || index >= l.size {
                return "", ErrIndex
        }
        n := l.head
        for i := 0; i < index; i++ {
                n = n.next
        }
        return n.value, nil
}

func (l *DoublyLinkedList) Front() (string, error) {
        if l.head == nil {
                return "", ErrEmpty
        }
        return l.head.value, nil
}

func (l *DoublyLinkedList) Back() (string, error) {
        if l.tail == nil {
                return "", ErrEmpty
        }
        return l.tail.value, nil
}

func (l *DoublyLinkedList) ToSlice() []string {
        out := make([]string, 0, l.size)
        for n := l.head; n != nil; n = n.next {
                out = append(out, n.value)
        }
        return out
}

// ToSliceReversed обходит список от хвоста к голове.
func (l *DoublyLinkedList) ToSliceReversed() []string {
        out := make([]string, 0, l.size)
        for n := l.tail; n != nil; n = n.prev {
                out = append(out, n.value)
        }
        return out
}

func (l *DoublyLinkedList) Append(v string) error {
        l.PushBack(v)
        return nil
}

// Djb2 - хеш-функция DJB2 (h = h*33 + c).
func Djb2(key string) uint64 {
        h := uint64(5381)
        for i := 0; i < len(key); i++ {
                h = h*33 + uint64(key[i])
        }
        return h
}

// HashTable - множество строк на цепочках; расширяется при заполнении > 0.75.
type HashTable struct {
        buckets []SinglyLinkedList
        size    int
}

// NewHashTable создаёт таблицу с заданным числом корзин (минимум 1).
func NewHashTable(buckets int) *HashTable {
        if buckets < 1 {
                buckets = 1
        }
        return &HashTable{buckets: make([]SinglyLinkedList, buckets)}
}

func (h *HashTable) TypeName() string { return "HashTable" }
func (h *HashTable) Size() int        { return h.size }
func (h *HashTable) BucketCount() int { return len(h.buckets) }

func (h *HashTable) idx(key string, n int) int { return int(Djb2(key) % uint64(n)) }

func (h *HashTable) Contains(key string) bool {
        return h.buckets[h.idx(key, len(h.buckets))].Contains(key)
}

// Insert возвращает false, если ключ уже есть.
func (h *HashTable) Insert(key string) bool {
        if h.Contains(key) {
                return false
        }
        if (h.size+1)*4 > len(h.buckets)*3 {
                h.rehash(len(h.buckets) * 2)
        }
        h.buckets[h.idx(key, len(h.buckets))].PushFront(key)
        h.size++
        return true
}

func (h *HashTable) Erase(key string) bool {
        if !h.buckets[h.idx(key, len(h.buckets))].RemoveByValue(key) {
                return false
        }
        h.size--
        return true
}

func (h *HashTable) rehash(n int) {
        fresh := make([]SinglyLinkedList, n)
        for i := range h.buckets {
                for _, k := range h.buckets[i].ToSlice() {
                        fresh[h.idx(k, n)].PushFront(k)
                }
        }
        h.buckets = fresh
}

// MaxChainLength - длина самой длинной цепочки.
func (h *HashTable) MaxChainLength() int {
        m := 0
        for i := range h.buckets {
                if s := h.buckets[i].Size(); s > m {
                        m = s
                }
        }
        return m
}

func (h *HashTable) ToSlice() []string {
        out := make([]string, 0, h.size)
        for i := range h.buckets {
                out = append(out, h.buckets[i].ToSlice()...)
        }
        return out
}

func (h *HashTable) Clear() {
        for i := range h.buckets {
                h.buckets[i].Clear()
        }
        h.size = 0
}

func (h *HashTable) Append(v string) error {
        h.Insert(v)
        return nil
}

type singlyNode struct {
        value string
        next  *singlyNode
}

// SinglyLinkedList - односвязный список с указателями на голову и хвост.
type SinglyLinkedList struct {
        head, tail *singlyNode
        size       int
}

func (l *SinglyLinkedList) TypeName() string { return "SinglyLinkedList" }
func (l *SinglyLinkedList) Size() int        { return l.size }
func (l *SinglyLinkedList) Clear()           { l.head, l.tail, l.size = nil, nil, 0 }

func (l *SinglyLinkedList) PushFront(v string) {
        n := &singlyNode{value: v, next: l.head}
        l.head = n
        if l.tail == nil {
                l.tail = n
        }
        l.size++
}

func (l *SinglyLinkedList) PushBack(v string) {
        n := &singlyNode{value: v}
        if l.tail == nil {
                l.head, l.tail = n, n
        } else {
                l.tail.next = n
                l.tail = n
        }
        l.size++
}

func (l *SinglyLinkedList) find(v string) *singlyNode {
        for n := l.head; n != nil; n = n.next {
                if n.value == v {
                        return n
                }
        }
        return nil
}

func (l *SinglyLinkedList) Contains(v string) bool { return l.find(v) != nil }

// InsertAfter вставляет value после первого вхождения target.
func (l *SinglyLinkedList) InsertAfter(target, value string) bool {
        t := l.find(target)
        if t == nil {
                return false
        }
        t.next = &singlyNode{value: value, next: t.next}
        if l.tail == t {
                l.tail = t.next
        }
        l.size++
        return true
}

// InsertBefore вставляет value перед первым вхождением target.
func (l *SinglyLinkedList) InsertBefore(target, value string) bool {
        if l.head == nil {
                return false
        }
        if l.head.value == target {
                l.PushFront(value)
                return true
        }
        for n := l.head; n.next != nil; n = n.next {
                if n.next.value == target {
                        n.next = &singlyNode{value: value, next: n.next}
                        l.size++
                        return true
                }
        }
        return false
}

func (l *SinglyLinkedList) PopFront() (string, error) {
        if l.head == nil {
                return "", ErrEmpty
        }
        v := l.head.value
        l.head = l.head.next
        if l.head == nil {
                l.tail = nil
        }
        l.size--
        return v, nil
}

func (l *SinglyLinkedList) PopBack() (string, error) {
        if l.head == nil {
                return "", ErrEmpty
        }
        if l.head == l.tail {
                return l.PopFront()
        }
        prev := l.head
        for prev.next != l.tail {
                prev = prev.next
        }
        v := l.tail.value
        prev.next = nil
        l.tail = prev
        l.size--
        return v, nil
}

// RemoveByValue удаляет первое вхождение value.
func (l *SinglyLinkedList) RemoveByValue(v string) bool {
        if l.head == nil {
                return false
        }
        if l.head.value == v {
                _, _ = l.PopFront()
                return true
        }
        for n := l.head; n.next != nil; n = n.next {
                if n.next.value == v {
                        victim := n.next
                        n.next = victim.next
                        if victim == l.tail {
                                l.tail = n
                        }
                        l.size--
                        return true
                }
        }
        return false
}

// RemoveAfter удаляет элемент, следующий за target.
func (l *SinglyLinkedList) RemoveAfter(target string) bool {
        t := l.find(target)
        if t == nil || t.next == nil {
                return false
        }
        victim := t.next
        t.next = victim.next
        if victim == l.tail {
                l.tail = t
        }
        l.size--
        return true
}

// RemoveBefore удаляет элемент, стоящий перед target.
func (l *SinglyLinkedList) RemoveBefore(target string) bool {
        if l.head == nil || l.head.value == target {
                return false
        }
        if l.head.next != nil && l.head.next.value == target {
                _, _ = l.PopFront()
                return true
        }
        for n := l.head; n.next != nil && n.next.next != nil; n = n.next {
                if n.next.next.value == target {
                        n.next = n.next.next
                        l.size--
                        return true
                }
        }
        return false
}

func (l *SinglyLinkedList) GetAt(index int) (string, error) {
        if index < 0 || index >= l.size {
                return "", ErrIndex
        }
        n := l.head
        for i := 0; i < index; i++ {
                n = n.next
        }
        return n.value, nil
}

func (l *SinglyLinkedList) Front() (string, error) {
        if l.head == nil {
                return "", ErrEmpty
        }
        return l.head.value, nil
}

func (l *SinglyLinkedList) Back() (string, error) {
        if l.tail == nil {
                return "", ErrEmpty
        }
        return l.tail.value, nil
}

func (l *SinglyLinkedList) ToSlice() []string {
        out := make([]string, 0, l.size)
        for n := l.head; n != nil; n = n.next {
                out = append(out, n.value)
        }
        return out
}

func (l *SinglyLinkedList) Append(v string) error {
        l.PushBack(v)
        return nil
}

// Stack - стек на основе односвязного списка (вершина - голова списка).
type Stack struct{ list SinglyLinkedList }

func (s *Stack) TypeName() string { return "Stack" }
func (s *Stack) Size() int        { return s.list.Size() }
func (s *Stack) Clear()           { s.list.Clear() }
func (s *Stack) Push(v string)    { s.list.PushFront(v) }
func (s *Stack) Pop() (string, error) {
        return s.list.PopFront()
}
func (s *Stack) Top() (string, error) { return s.list.Front() }

// ToSlice возвращает элементы от дна к вершине.
func (s *Stack) ToSlice() []string {
        items := s.list.ToSlice()
        for i, j := 0, len(items)-1; i < j; i, j = i+1, j-1 {
                items[i], items[j] = items[j], items[i]
        }
        return items
}

func (s *Stack) Append(v string) error {
        s.Push(v)
        return nil
}

// Queue - очередь FIFO на односвязном списке с хвостом: Push и Pop за O(1).
type Queue struct{ list SinglyLinkedList }

func (q *Queue) Clear()                 { q.list.Clear() }
func (q *Queue) Push(v string)          { q.list.PushBack(v) }

package ds

import (
        "fmt"
        "strconv"
)

type treeNode struct {
        value       int
        left, right *treeNode
}

// CompleteBinaryTree - полное двоичное дерево: уровни заполняются слева направо.
type CompleteBinaryTree struct {
        root *treeNode
        size int
}

func (t *CompleteBinaryTree) TypeName() string { return "CompleteBinaryTree" }
func (t *CompleteBinaryTree) Size() int        { return t.size }
func (t *CompleteBinaryTree) Clear()           { t.root, t.size = nil, 0 }

// Insert добавляет значение в первую свободную позицию (обход в ширину).
func (t *CompleteBinaryTree) Insert(v int) {
        n := &treeNode{value: v}
        t.size++
        if t.root == nil {
                t.root = n
                return
        }
        queue := []*treeNode{t.root}
        for {
                cur := queue[0]
                queue = queue[1:]
                if cur.left == nil {
                        cur.left = n
                        return
                }
                if cur.right == nil {
                        cur.right = n
                        return
                }
                queue = append(queue, cur.left, cur.right)
        }
}

func (t *CompleteBinaryTree) Find(v int) bool {
        for _, x := range t.LevelOrder() {
                if x == v {
                        return true
                }
        }
        return false
}

// IsComplete проверяет, что все уровни, кроме последнего, заполнены, а последний - слева направо.
func (t *CompleteBinaryTree) IsComplete() bool {
        queue := []*treeNode{t.root}
        seenNil := false
        for len(queue) > 0 {
                cur := queue[0]
                queue = queue[1:]
                if cur == nil {
                        seenNil = true
                        continue
                }
                if seenNil {
                        return false
                }
                queue = append(queue, cur.left, cur.right)
        }
        return true
}

func height(n *treeNode) int {
        if n == nil {
                return -1
        }
        l, r := height(n.left), height(n.right)
        if r > l {
                l = r
        }
        return l + 1
}

// Height - высота дерева (-1 для пустого).
func (t *CompleteBinaryTree) Height() int { return height(t.root) }

func (t *CompleteBinaryTree) LevelOrder() []int {
        var out []int
        if t.root == nil {
                return out
        }
        queue := []*treeNode{t.root}
        for len(queue) > 0 {
                cur := queue[0]
                queue = queue[1:]
                out = append(out, cur.value)
                if cur.left != nil {
                        queue = append(queue, cur.left)
                }
                if cur.right != nil {
                        queue = append(queue, cur.right)
                }
        }
        return out
}

func walk(n *treeNode, order int, out *[]int) {
        if n == nil {
                return
        }
        if order == 0 { // pre
                *out = append(*out, n.value)
        }
        walk(n.left, order, out)
        if order == 1 { // in
                *out = append(*out, n.value)
        }
        walk(n.right, order, out)
        if order == 2 { // post
                *out = append(*out, n.value)
        }
}

func (t *CompleteBinaryTree) PreOrder() []int  { var o []int; walk(t.root, 0, &o); return o }
func (t *CompleteBinaryTree) InOrder() []int   { var o []int; walk(t.root, 1, &o); return o }
func (t *CompleteBinaryTree) PostOrder() []int { var o []int; walk(t.root, 2, &o); return o }

func (t *CompleteBinaryTree) ToSlice() []string {
        vals := t.LevelOrder()
        out := make([]string, len(vals))
        for i, v := range vals {
                out[i] = strconv.Itoa(v)
        }
        return out
}

func (t *CompleteBinaryTree) Append(s string) error {
        v, err := strconv.Atoi(s)
        if err != nil {
                return fmt.Errorf("ds: not an integer: %q", s)
        }
        t.Insert(v)
        return nil
}
